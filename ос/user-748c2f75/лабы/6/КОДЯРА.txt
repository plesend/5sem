#include <windows.h>
#include <stdio.h>

#define MUTEX L"Global\\mutex"

int main()
{
	HANDLE mutex = OpenMutexW(MUTEX_ALL_ACCESS, FALSE, MUTEX);
	if (!mutex) {
		printf("cant open mutex :(((((((");
		return 1;
	}

	const char array[13] = {
	'u', 's', 'e', 'r', '-',
	'7', '4', '8', 'c',
	'2', 'f', '7', '5'
	};

	for (int i = 0; i <= 90; i++) {
		int lock = (i >= 30 && i <= 60);

		if (lock) WaitForSingleObject(mutex, INFINITE);

		int index = i % 13;
		char symb = array[index];
		printf("B iteration %d char: %c\n", i, symb);

		if (lock) ReleaseMutex(mutex);

		Sleep(100);
	}

	CloseHandle(mutex);
	return 0;
}
