dir="$1"

if [ ! -d "$dir" ]; then
    echo "Directory does not exist: $dir"
    exit 1

format_size() {
	local size=$1
    	local unit="B"

	if [ "$size" -ge 1024 ]; then
        	size=$(awk "BEGIN {printf \"%.2f\", $size/1024}")
        	unit="KB"
    	fi
    	if (( $(echo "$size >= 1024" | bc -l) )); then
        	size=$(awk "BEGIN {printf \"%.2f\", $size/1024}")
        	unit="MB"
    	fi
    	if (( $(echo "$size >= 1024" | bc -l) )); then
        	size=$(awk "BEGIN {printf \"%.2f\", $size/1024}")
        	unit="GB"
    	fi
    	echo "$size $unit"
}

for entry in "$dir"/*; do
    if [ -d "$entry" ]; then
        count=$(find "$entry" -type f | wc -l)
        echo "Directory: $(basename "$entry") — $count files"
    elif [ -f "$entry" ]; then
        size_bytes=$(stat -c %s "$entry")
        size_formatted=$(format_size "$size_bytes")
        echo "File: $(basename "$entry") — $size_formatted"
    fi
done
