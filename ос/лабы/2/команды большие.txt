@echo off
setlocal

rem Проверяем, задан ли первый параметр (режим работы)
if "%1"=="" (
    echo Usage: bat_05 {create/delete} {filename}
    goto :eof
)

rem Проверяем, задан ли второй параметр (имя файла)
if "%2"=="" (
    echo Please specify a filename.
    goto :eof
)

set MODE=%1
set FILENAME=%2

if /i "%MODE%"=="create" (
    if exist "%FILENAME%" (
        echo File "%FILENAME%" already exists.
    ) else (
        echo.>"%FILENAME%"
        echo File "%FILENAME%" has been created.
    )
) else if /i "%MODE%"=="delete" (
    if exist "%FILENAME%" (
        del "%FILENAME%"
        echo File "%FILENAME%" has been deleted.
    ) else (
        echo File "%FILENAME%" does not exist.
    )
) else (
    echo Invalid mode. Use "create" or "delete".
)

pause
