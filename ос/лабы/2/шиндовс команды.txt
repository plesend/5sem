function Read-Number($prompt) {
    while ($true) {
        $input = Read-Host $prompt
        if ($input -match '^-?\d+(\.\d+)?$') {
            return [double]$input
        }
        else {
            Write-Host "Введите число!" -ForegroundColor Red
        }
    }
}

function ifC($number, $inputStr) {
    Write-Host "$inputStr = $($number*1.8+32)F = $($number+273.15)K"
}

function ifF($number, $inputStr) {
    Write-Host "$inputStr = $(($number-32)*5/9)C = $(($number-32)*5/9+273.15)K"
}

function ifK($number, $inputStr) {
    Write-Host "$inputStr = $($number-273.15)C = $(($number-273.15)*9/5+32)F"
}

while ($true) {
    $userInput = Read-Host "Введите значение (например 100C, 212F, 300K)"

    if ([string]::IsNullOrEmpty($userInput)) {
        Write-Host "Вы ничего не ввели!" -ForegroundColor Red
        continue
    }

    $letter = $userInput[-1]
    $letter = [string]$letter

    if ($letter -match '[A-Za-z]') {
        $number = [double]$userInput.Substring(0, $userInput.Length - 1)
    } elseif ($letter -match '\d') {
        $number = [double]$userInput
        $letter = ''
    } else {
        Write-Host "Введите нормальное значение" -ForegroundColor Red
        continue
    }

    switch ($letter.ToUpper()) {
        'C' { ifC $number $userInput }
        ''  { ifC $number $userInput }
        'F' { ifF $number $userInput }
        'K' { ifK $number $userInput }
        default { Write-Host "Invalid choice!" -ForegroundColor Red }
    }
}
